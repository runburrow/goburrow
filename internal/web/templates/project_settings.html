Quick setup

{{$dsn := .Data.DSN}} {{if eq $p.Platform "python"}} 
pip install sentry-sdk

import sentry_sdk
sentry_sdk.init(dsn="{{$dsn}}", environment="production")
{{else if eq $p.Platform "node"}} 
npm install @sentry/node

const Sentry = require("@sentry/node");
Sentry.init({ dsn: "{{$dsn}}", environment: "production" });
{{else if eq $p.Platform "javascript"}} 
npm install @sentry/browser

import * as Sentry from "@sentry/browser";
Sentry.init({ dsn: "{{$dsn}}", environment: "production" });
{{else if eq $p.Platform "go"}} 
go get github.com/getsentry/sentry-go

err := sentry.Init(sentry.ClientOptions{Dsn: "{{$dsn}}"})
defer sentry.Flush(2 * time.Second)
{{else if eq $p.Platform "ruby"}} 
gem install sentry-ruby

Sentry.init { |config| config.dsn = "{{$dsn}}" }
{{else if eq $p.Platform "php"}} 
composer require sentry/sentry

\Sentry\init(['dsn' => '{{$dsn}}']);
{{else if eq $p.Platform "rust"}} 
cargo add sentry

let _guard = sentry::init(("{{$dsn}}", sentry::ClientOptions::default()));
{{else if eq $p.Platform "java"}} 
// io.sentry:sentry
Sentry.init(options -> options.setDsn("{{$dsn}}"));
{{else if eq $p.Platform "dotnet"}} 
dotnet add package Sentry

SentrySdk.Init(o => o.Dsn = "{{$dsn}}");
{{else}}

Install the Sentry SDK for your language and initialize it with the DSN above. Burrow speaks the same ingestion protocol.

{{end}}

Tip: set traces_sample_rate to 0. Burrow currently accepts and discards performance data.
